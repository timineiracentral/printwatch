    ```bash
    #!/bin/bash
    set -euo pipefail

    QUICK=false
    if [[ "${1:-}" == "--quick" ]]; then QUICK=true; fi

    PASS=0; FAIL=0; WARN=0
    pass()  { echo "  ✓ $*";  PASS=$((PASS+1)); }
    fail()  { echo "  ✗ $*" >&2; FAIL=$((FAIL+1)); }
    warn()  { echo "  ⚠ $*"; WARN=$((WARN+1)); }
    check() { local label="$1"; shift; if "$@" >/dev/null 2>&1; then pass "$label"; else fail "$label"; fi; }

    BASE_URL="${BASE_URL:-http://localhost:8000}"

    echo "PrintWatch — validate-phase3 ($([ "$QUICK" = "true" ] && echo "quick" || echo "full"))"
    echo

    # 1. Container backend up
    check "01 backend container running" docker compose ps backend | grep -q "running\|Up"

    # 2. OpenAPI: title = PrintWatch
    check "02 GET /api/v1/openapi.json info.title == 'PrintWatch'" bash -c "curl -fsS $BASE_URL/api/v1/openapi.json | python -c \"import json,sys; assert json.load(sys.stdin)['info']['title']=='PrintWatch'\""

    # 3. /healthz status=ok
    check "03 GET /healthz status='ok'" bash -c "curl -fsS $BASE_URL/healthz | python -c \"import json,sys; assert json.load(sys.stdin)['status']=='ok'\""

    # 4. /api/v1/health 200 + db_reachable=True
    check "04 GET /api/v1/health db_reachable=true" bash -c "curl -fsS $BASE_URL/api/v1/health | python -c \"import json,sys; j=json.load(sys.stdin); assert j['db_reachable']==True\""

    # 5. /api/v1/jobs shape
    check "05 GET /api/v1/jobs shape items/total/page/size" bash -c "curl -fsS '$BASE_URL/api/v1/jobs?page=1&size=10' | python -c \"import json,sys; j=json.load(sys.stdin); assert {'items','total','page','size'} <= set(j.keys())\""

    # 6. Indices existem
    check "06 4 índices SQLite existem em print_jobs" bash -c "docker compose exec backend python -c \"from sqlalchemy import inspect; from app.db.session import engine; idx={i['name'] for i in inspect(engine).get_indexes('print_jobs')}; assert {'idx_print_jobs_timestamp','idx_print_jobs_username_timestamp','idx_print_jobs_printer_timestamp','idx_print_jobs_job_id'} <= idx\""

    # 7. EXPLAIN QUERY PLAN usa índice em timestamp
    check "07 EXPLAIN QUERY PLAN usa idx_print_jobs_timestamp" bash -c "docker compose exec backend python -c \"from sqlalchemy import text; from app.db.session import engine; r=engine.connect().execute(text('EXPLAIN QUERY PLAN SELECT * FROM print_jobs ORDER BY timestamp DESC LIMIT 50')).fetchall(); print(r); assert any('idx_print_jobs_timestamp' in str(row) for row in r)\""

    # 8. /api/v1/export/csv headers
    check "08 Content-Type text/csv + Content-Disposition attachment" bash -c "curl -sI '$BASE_URL/api/v1/export/csv' | grep -E '^Content-Type: text/csv' && curl -sI '$BASE_URL/api/v1/export/csv' | grep -E '^Content-Disposition: attachment'"

    # 9. CSV BOM UTF-8
    TMPCSV=$(mktemp)
    curl -fsS "$BASE_URL/api/v1/export/csv" -o "$TMPCSV"
    check "09 CSV começa com BOM UTF-8 (ef bb bf)" bash -c "xxd \"$TMPCSV\" | head -1 | grep -q 'efbb bf'"

    # 10. CSV separador ;
    check "10 CSV header usa separador ';'" bash -c "head -1 \"$TMPCSV\" | grep -q 'Data/Hora;Usuário;Impressora'"
    rm -f "$TMPCSV"

    # 11. /jobs?date_from>date_to → 422
    check "11 GET /jobs date_from > date_to → 422" bash -c "[ \"\$(curl -sw '%{http_code}' -o /dev/null '$BASE_URL/api/v1/jobs?date_from=2026-12-31&date_to=2026-01-01')\" = '422' ]"

    # 12. /jobs?size=999 → 422
    check "12 GET /jobs size>500 → 422" bash -c "[ \"\$(curl -sw '%{http_code}' -o /dev/null '$BASE_URL/api/v1/jobs?size=999')\" = '422' ]"

    # 13. /printers ordenado alfabético
    check "13 GET /printers retorna lista JSON" bash -c "curl -fsS $BASE_URL/api/v1/printers | python -c \"import json,sys; v=json.load(sys.stdin); assert isinstance(v, list); assert v == sorted(v), f'not sorted: {v}'\""

    # 14. CORS preflight com origem permitida
    check "14 CORS preflight allow-origin para localhost:5173" bash -c "curl -sI -X OPTIONS -H 'Origin: http://localhost:5173' -H 'Access-Control-Request-Method: GET' $BASE_URL/api/v1/jobs | grep -E '^access-control-allow-origin: http://localhost:5173'"

    # 15. CORS preflight origem não permitida não retorna header
    check "15 CORS bloqueia origem evil.com" bash -c "! curl -sI -X OPTIONS -H 'Origin: http://evil.com' -H 'Access-Control-Request-Method: GET' $BASE_URL/api/v1/jobs | grep -qE '^access-control-allow-origin: http://evil.com'"

    # 16. pytest dentro do container
    check "16 pytest suite green" bash -c "docker compose exec -T backend pytest -q"

    echo
    echo "Resumo (auto): $PASS pass / $FAIL fail / $WARN warn"

    # Modo completo: checkpoint humano
    if [[ "$QUICK" == "false" ]]; then
      echo
      echo "=== 17. CHECKPOINT HUMANO ==="
      echo "Pré-requisitos: PC Windows com impressora IPP configurada apontando para a VM VM_HOST."
      echo
      echo "Passos manuais:"
      echo "  a) Em um PC Windows: imprimir um documento de teste para a impressora PrintWatch."
      echo "  b) Aguardar até 30 segundos."
      echo "  c) Em outro terminal: curl 'http://VM_HOST:8000/api/v1/jobs?username=<seu-usuario>' | head -50"
      echo "  d) Confirmar que o job aparece com os campos preenchidos (printer, pages, timestamp, host_origin)."
      echo "  e) Baixar CSV: curl 'http://VM_HOST:8000/api/v1/export/csv?username=<seu-usuario>' -o /tmp/test.csv"
      echo "  f) Abrir /tmp/test.csv no Excel pt-BR — confirmar que abre SEM prompt de import, com acentos corretos e colunas separadas."
      echo
      read -r -p "Checkpoint #17 passou? [y/N]: " ans
      if [[ "${ans,,}" == "y" ]]; then
        pass "17 checkpoint humano aprovado"
      else
        fail "17 checkpoint humano NÃO aprovado — registrar evidência do que falhou em 03-VERIFICATION.md"
      fi
      echo
      echo "Resumo final: $PASS pass / $FAIL fail / $WARN warn"
    fi

    if [[ $FAIL -gt 0 ]]; then exit 1; fi
    exit 0
    ```
